## 1. Питч

**«Pinball Odyssey: 8 Realms»** — вертикальный аркадный пинболл для одной руки. 88 рукотворных столов, 8 живых биомов от Пустыни до Космоса. Тапнул левую половину экрана — левый флиппер бьёт. Правую — правый. Не нужно целиться, не нужно считать. Только чистая физика, неоновые взрывы, тряска камеры и дофамин от комбо-ударов. ## 2. GDD

### 2.1 Core Loop
1. **Запуск мяча** — тап по экрану в зоне «пружины» (или авто-запуск через 3 секунды).
2. **Игра** — мяч летит по столу, отскакивает от бамперов, собирает бонусы, взаимодействует с механиками биома.
3. **Потеря мяча** — мяч падает мимо флипперов в «Dead Zone».
4. **Условие победы** — набрать целевой счёт (Target Score) за выделенное количество мячей (обычно 3).
5. **Между уровнями** — экран с выбором 1 из 3 случайных бонусов (Power-up) на следующий уровень: «+1 мяч», «Усиленные флипперы», «Магнит под центром».

### 2.3 Система звёзд
- ⭐ — пройти уровень (достичь Target Score).
- ⭐⭐ — набрать средний порог.
- ⭐⭐⭐ — набрать высший порог (требует комбо-мастерства).
- Звёзды открывают следующие биомы (например, доступ к Лесу — 15 звёзд в Пустыне).

## 3. Архитектура Godot 4

### 3.1 Дерево сцен (`table.tscn`)
```
Node2D (TableManager)
├── Background (Sprite2D / ParallaxBackground)
├── Walls (StaticBody2D) — полигон стола
├── LaunchLane (Area2D) — зона запуска
├── BallSpawn (Marker2D)
├── BallContainer (Node2D)
├── Flippers (Node2D)
│   ├── LeftFlipper (Flipper)
│   └── RightFlipper (Flipper)
├── Bumpers (Node2D)
│   └── Bumper1..N (Bumper)
├── Hazards (Node2D) — движущиеся препятствия биома
├── DeadZone (Area2D) — за флипперами
├── ParticlesContainer (Node2D)
└── Camera2D
```

### 3.2 Singletons (AutoLoad)
- `GameManager` — прогресс, загрузка уровней, звёзды, сохранения.
- `FXManager` — тряска камеры, частицы, вспышки.
- `AudioManager` — SFX и музыка биомов.

### 3.3 Data-Driven уровни
Каждый уровень — это `LevelData` (Resource `.tres`), ссылающийся на `BiomeData`. Дизайнер создаёт 88 `.tres` файлов (или генерирует скриптом), не леза в код.

### 4.3 `ball.gd` — Физика мяча
```gdscript
class_name Ball
extends RigidBody2D

@export var max_speed: float = 2500.0
@export var min_speed: float = 80.0
@export var trail_length: int = 12

@onready var sprite: Sprite2D = $Sprite2D
@onready var trail: Line2D = $Trail

var trail_points: Array[Vector2] = []

func _ready():
	contact_monitor = true
	max_contacts_reported = 5
	# Включаем CCD в Inspector, чтобы не проходить сквозь флипперы
	body_entered.connect(_on_body_entered)

func _integrate_forces(state: PhysicsDirectBodyState2D):
	# Кап скорости — иначе вылетит за стены на слабом телефоне
	var spd = state.linear_velocity.length()
	if spd > max_speed:
		state.linear_velocity = state.linear_velocity.normalized() * max_speed
	elif spd < min_speed and spd > 0.1:
		state.linear_velocity = state.linear_velocity.normalized() * min_speed
	
	# Обновление следа
	trail_points.push_front(global_position)
	if trail_points.size() > trail_length:
		trail_points.resize(trail_length)
	trail.clear_points()
	for p in trail_points:
		trail.add_point(to_local(p))

func _on_body_entered(body: Node):
	if body.is_in_group("bumpers"):
		FXManager.play_bumper_hit(global_position)
	elif body.is_in_group("flippers"):
		FXManager.play_flipper_hit(global_position)

func apply_biome_physics(g: float, ld: float, ad: float):
	gravity_scale = g
	linear_damp = ld
	angular_damp = ad
```

### 4.4 `flipper.gd` — Флиппер (AnimatableBody2D)
```gdscript
class_name Flipper
extends AnimatableBody2D

enum Side { LEFT, RIGHT }

@export var side: Side = Side.LEFT
@export var rest_angle: float = 25.0
@export var active_angle: float = -35.0
@export var speed: float = 18.0

var is_active: bool = false
var current_angle: float = 0.0

func _ready():
	current_angle = rest_angle
	rotation_degrees = rest_angle
	# Для правого флиппера углы обычно зеркальны; настраиваются в инспекторе

func _physics_process(delta):
	var target = active_angle if is_active else rest_angle
	current_angle = lerp(current_angle, target, speed * delta)
	rotation_degrees = current_angle

func activate():
	is_active = true

func deactivate():
	is_active = false
```

### 4.5 `bumper.gd` — Бампер с анимацией удара
```gdscript
class_name Bumper
extends Area2D

@export var base_score: int = 100
@export var impulse_force: float = 600.0
@export var hit_flash_time: float = 0.08

@onready var sprite: Sprite2D = $Sprite2D
var original_scale: Vector2

func _ready():
	original_scale = sprite.scale
	body_entered.connect(_on_body_entered)

func _on_body_entered(body: Node):
	if not body is Ball:
		return
	var ball := body as Ball
	
	# Отскок в направлении от центра бампера
	var dir := (ball.global_position - global_position).normalized()
	ball.apply_central_impulse(dir * impulse_force * GameManager.current_bumper_mult)
	
	# Очки
	GameManager.add_score(base_score)
	
	# FX
	FXManager.play_bumper_hit(global_position)
	_animate_hit()

func _animate_hit():
	sprite.scale = original_scale * 1.4
	var tween := create_tween().set_trans(Tween.TRANS_ELASTIC).set_ease(Tween.EASE_OUT)
	tween.tween_property(sprite, "scale", original_scale, 0.25)
	
	sprite.modulate = Color.YELLOW
	await get_tree().create_timer(hit_flash_time).timeout
	sprite.modulate = Color.WHITE
```

### 4.6 `fx_manager.gd` — Тряска, частицы, вспышки
```gdscript
extends Node

@onready var camera: Camera2D
@onready var particles_root: Node2D

var shake_strength: float = 0.0
var shake_decay: float = 8.0

func _ready():
	if instance != null:
		push_warning("FXManager duplicate")
	instance = self

func _process(delta):
	if shake_strength > 0.0:
		camera.offset = Vector2(
			randf_range(-shake_strength, shake_strength),
			randf_range(-shake_strength, shake_strength)
		)
		shake_strength = maxf(0.0, shake_strength - shake_decay * delta)
	else:
		camera.offset = Vector2.ZERO

func play_bumper_hit(pos: Vector2):
	shake_strength = 10.0
	_spawn_at(pos, preload("res://fx/bumper_burst.tscn"))

func play_flipper_hit(pos: Vector2):
	shake_strength = 4.0

func play_ball_lost():
	shake_strength = 6.0

func play_level_complete():
	shake_strength = 20.0
	_spawn_at(Vector2.ZERO, preload("res://fx/confetti.tscn"))

func _spawn_at(pos: Vector2, scene: PackedScene):
	var node := scene.instantiate()
	node.global_position = pos
	particles_root.add_child(node)

static var instance: FXManager
static func play_bumper_hit(pos: Vector2): instance.play_bumper_hit(pos)
static func play_flipper_hit(pos: Vector2): instance.play_flipper_hit(pos)
static func play_ball_lost(): instance.play_ball_lost()
static func play_level_complete(): instance.play_level_complete()
```

### 4.7 `table_manager.gd` — Управление столом
```gdscript
extends Node2D

@export var ball_scene: PackedScene

@onready var ball_spawn: Marker2D = $BallSpawn
@onready var ball_container: Node2D = $BallContainer
@onready var flippers: Node2D = $Flippers
@onready var dead_zone: Area2D = $DeadZone

var current_ball: Ball
var balls_left: int = 3
var level_score: int = 0

func _ready():
	dead_zone.body_entered.connect(_on_ball_lost)
	_initialize_level(GameManager.current_level_data)

func _initialize_level(level: LevelData):
	if level == null:
		push_error("No level data!")
		return
	
	# Применяем биом
	if level.biome:
		$Background.texture = level.biome.background_texture
		$Walls.modulate = level.biome.wall_color
		GameManager.current_bumper_mult = level.biome.bumper_impulse_mult
	
	balls_left = level.max_balls
	level_score = 0
	GameManager.reset_level_score()
	_spawn_ball()

func _spawn_ball():
	if is_instance_valid(current_ball):
		current_ball.queue_free()
	
	current_ball = ball_scene.instantiate() as Ball
	current_ball.global_position = ball_spawn.global_position
	ball_container.add_child(current_ball)
	
	var bio = GameManager.current_level_data.biome
	if bio:
		current_ball.apply_biome_physics(bio.gravity_scale, bio.linear_damp, bio.angular_damp)

func _on_ball_lost(body: Node):
	if body != current_ball:
		return
	
	balls_left -= 1
	FXManager.play_ball_lost()
	
	if balls_left > 0:
		await get_tree().create_timer(1.2).timeout
		_spawn_ball()
	else:
		_end_level(false)

func _end_level(won: bool):
	set_process_input(false)
	if won:
		GameManager.on_level_completed(level_score)
	else:
		GameManager.on_game_over(level_score)

func _input(event):
	# Поддержка и мыши (для теста), и тачскрина
	var is_press := false
	var is_release := false
	var pos_x := 0.0
	
	if event is InputEventScreenTouch:
		is_press = event.pressed
		is_release = not event.pressed
		pos_x = event.position.x
	elif event is InputEventMouseButton:
		is_press = event.pressed and event.button_index == MOUSE_BUTTON_LEFT
		is_release = not event.pressed and event.button_index == MOUSE_BUTTON_LEFT
		pos_x = event.position.x
	
	if not is_press and not is_release:
		return
	
	var half := get_viewport().get_visible_rect().size.x * 0.5
	var left_flip := flippers.get_child(0) as Flipper
	var right_flip := flippers.get_child(1) as Flipper
	
	if pos_x < half:
		left_flip.activate() if is_press else left_flip.deactivate()
	else:
		right_flip.activate() if is_press else right_flip.deactivate()

func add_score(points: int):
	level_score += points
	GameManager.add_score(points)
	_check_win()

func _check_win():
	if level_score >= GameManager.current_level_data.target_score:
		_end_level(true)
```

### 4.8 `game_manager.gd` — Синглтон прогресса
```gdscript
extends Node

enum State { MENU, PLAYING, LEVEL_COMPLETE, GAME_OVER }
var current_state: State = State.MENU

var current_level_data: LevelData
var current_level_index: int = 0
var current_bumper_mult: float = 1.0

var level_score: int = 0
var total_score: int = 0
var stars_per_level: Dictionary = {} # int -> int
var unlocked_biome: int = 0

const LEVEL_PATH := "res://levels/"
const SAVE_PATH := "user://save_v1.dat"

func _ready():
	_load_progress()

func start_level(index: int):
	current_level_index = index
	current_level_data = _load_level_resource(index)
	current_state = State.PLAYING
	level_score = 0
	get_tree().change_scene_to_file("res://scenes/table.tscn")

func add_score(points: int):
	level_score += points
	total_score += points

func reset_level_score():
	level_score = 0

func on_level_completed(final_score: int):
	current_state = State.LEVEL_COMPLETE
	var stars := _calc_stars(final_score)
	var id := current_level_data.level_id
	stars_per_level[id] = maxi(stars, stars_per_level.get(id, 0))
	_save_progress()
	# Переход на экран наград
	get_tree().change_scene_to_file("res://ui/level_complete.tscn")

func on_game_over(final_score: int):
	current_state = State.GAME_OVER
	_save_progress()
	get_tree().change_scene_to_file("res://ui/game_over.tscn")

func _calc_stars(score: int) -> int:
	var s := 0
	for thr in current_level_data.star_thresholds:
		if score >= thr:
			s += 1
	return clampi(s, 0, 3)

func _load_level_resource(index: int) -> LevelData:
	# В реальном проекте — загрузка из массива или по имени файла
	var path := LEVEL_PATH + "level_%03d.tres" % (index + 1)
	if ResourceLoader.exists(path):
		return load(path) as LevelData
	push_error("Level not found: " + path)
	return null

func _save_progress():
	var file := FileAccess.open(SAVE_PATH, FileAccess.WRITE)
	var data := {
		"stars": stars_per_level,
		"total": total_score,
		"last_level": current_level_index,
		"unlocked_biome": unlocked_biome
	}
	file.store_var(data)

func _load_progress():
	if not FileAccess.file_exists(SAVE_PATH):
		return
	var file := FileAccess.open(SAVE_PATH, FileAccess.READ)
	var data: Dictionary = file.get_var()
	if data.has("stars"):
		stars_per_level = data["stars"]
	if data.has("total"):
		total_score = data["total"]
	if data.has("unlocked_biome"):
		unlocked_biome = data["unlocked_biome"]
```

## 5. Как делать 88 уровней не вручную

Создайте **8 базовых сцен столов** (`table_desert_base.tscn`, `table_forest_base.tscn`…). В каждой — расставлены стены и флипперы, но бамперы и препятствия — через скрипт или `Editable Children`.

Для 11 вариантов внутри биома используйте **LevelData** с разными:
- `target_score` (растёт экспоненциально: 5k → 8k → 12k…)
- `star_thresholds`
- `special_rules` (на уровне 11 — `moving_bumpers`, на уровне 5 — `multiball`)
- Разные `table_scene` (11 вариантов layout'ов на биом).

Так дизайнер делает 8×3 = 24 уникальных layout'а, а остальные 64 — вариации через параметры `.tres`.

## Почему Godot, а не Three.js

**Three.js** — это 3D-движок для браузера. Для 2D-пинболла это как ездить на танке в магазин за хлебом. 

Проблема не в Three.js как таковом, а в **доставке на телефон**. Чтобы игра с Three.js появилась в RuStore, её нужно завернуть в WebView (Capacitor, Cordova, Tauri). И тут начинается:

- **Физика** — либо пишешь сам (чистый ад), либо тащишь Matter.js/Ammo.js, который жрёт батарею.
- **Производительность** — на бюджетных Android 9–11 WebView лагает при 60fps и физике. Пинболл без плавности — это не пинболл, это мучение.
- **Нативные фичи** — пуши, реклама (MyTarget/Yandex), покупки, сохранения в облако. В WebView это костыли через JS-bridge.
- **Размер** — казалось бы, веб лёгкий. Но WebView-обёртка + Three.js + физика + аудио = 30–50 МБ. Godot 4 даёт нативный APK в 12–18 МБ с физикой, рендерером и аудио из коробки.

**Three.js крут, если:** ты делаешь Telegram WebApp, VK Mini Apps или браузерную демку. Для нативного RuStore — это стрельба по воробьям из пушки.

**Godot** — потому что там `RigidBody2D`, `Area2D`, флипперы и бамперы работают из коробки. Ты не пишешь физику — ты расставляешь объекты. APK маленький, работает на Android 8+, билдится в один клик.

## Артефакты и Музей — это 🔥

Идея с музеем, как в Dragonborn DLC (дом в Солстхейме, где выставляешь уникальные предметы) — **это именно то, чего не хватает мобильному пинболлу**. В обычном пинболле нет retention'а: поиграл 5 минут — забыл. А коллекционка цепляет на месяцы.

Вот как это можно вплести без слома баланса:

### 1. Артефакты — чисто косметика + лор
Они **не дают статов** в игре. Не «+10% к скорости мяча». Это важно — иначе сломается матчмейкинг (если он будет) или баланс уровней. Артефакты — это **трофеи**:

- **Выпадают** с вероятностью 0.5–3% при прохождении уровня.

## 2. Философия игры: «Смотритель Машин»

Давай придумаем, **почему** игрок бьёт мячиком по столу и почему оттуда выпадают артефакты.

**Почему это работает:**
- Пинболл — это ритуал, а не аркада. Каждый удар — это не «набей очки», а «почини мир».
- Артефакты — не случайный лут, а **куски реальности**, которые ты выбил из сломанного механизма.
- Музей — это не просто коллекция. Это **Зал Восстановления**, где собранные артефакты начинают взаимодействовать друг с другом, открывая лор и давая бонусы.

## 3. Магия игры: что должно чувствоваться

Я предлагаю зафиксировать три эмоциональных столпа:

### A. Вес и Мощь
Мяч не должен быть «прыгучей точкой». Он должен ощущаться как **стальной шарик массой 80 грамм**. Когда флиппер его подхватывает — должен быть звук удара металла о металл. Когда мяч врезается в бампер — экран не просто трясётся, а **отскакивает** от импульса. Игрок должен чувствовать, что он управляет **физическим объектом**, а не спрайтом.

### B. Драма Падения
Когда мяч катится к флипперам, должно быть **кино**. Камера слегка приближается, звук глушится, время замедляется на доли секунды. Игрок тапает — флиппер выстреливает, мяч улетает вверх, и время возвращается к норме. Это «bullet time» только в ключевых моментах, не надоедает, но даёт адреналин.

### C. Момент Артефакта
Когда выпадает редкий артефакт — игра **останавливается**. Мяч замирает в воздухе. Экран темнеет по краям. Артефакт медленно всплывает из стола, вращаясь, с лучами света. Звук — не «динь-динь», а **оркестровый удар** (как в *Diablo*, когда падает легендарка). Игрок должен хотеть пережить этот момент снова.

## 4. «AAA-физика» для пинболла — что это значит

Ты сказал «физика уровня AAA». Давай честно: мы не делаем *Red Dead Redemption 3*. Но для пинболла AAA-физика — это **не графика**, а **поведение**. Вот конкретика:

| Элемент | «Китайский клон» | «Наша игра» |
|---------|------------------|-------------|
| **Мяч** | Скользит как ледяная капля, без инерции | Имеет массу. Скатывается с ускорением. Отскакивает под углом, который можно предсказать |
| **Флипперы** | Мгновенно поворачиваются на 45° | Ускорение + торможение. Если тапнуть в момент контакта — мяч летит выше. Если рано — мяч соскальзывает |
| **Стены** | Одна текстура, один отскок | Разные материалы: дерево (глухой удар, слабый отскок), металл (звон, средний), резина (высокий отскок, тихий) |

| **Гравитация** | Постоянная 9.8 | Меняется по биому: в Пустыне — сухая, быстрая; в Болоте — вязкая, мяч тонет медленно; в Космосе — нулевая, мяч летит по инерции |

**Главное:** физика должна быть **честной**. Игрок, потратив 10 часов, должен интуитивно чувствовать: «Если я ударю сейчас, мяч полетит туда». Это мастерство, а не рандом.

## 5. Артефакты с бонусами: как не сломать игру

Ты сказал — артефакты дают бонусы. Окей, но давай сделаем это **интересно**, а не «+10% к очкам». Вот философия:

**Бонусы = Новые Правила, а не Статистика.**

Примеры:

- **«Сердце Магмы»** (из вулкана): Когда мяч касается трёх бамперов подряд, он становится «горячим» — оставляет огненный след, и следующий бампер даёт x3 очков. Но если мяч упадёт в dead zone горячим — бонус сгорает. **Риск/Награда.**
- **«Ледяная Слеза»** (из гор): Флипперы двигаются медленнее, но отскок от них точнее. Для игроков, которые любят контроль, а не хаос.
- **«Песочные Часы»** (из пустыни): Раз в уровень можно «заморозить» мяч на 2 секунды (тапнуть по мячу). Спасти от падения или выставить идеальный удар.
- **«Корень Миревуда»** (из болота): Бамперы иногда «вырастают» из стола в случайных местах на 5 секунд. Хаос, но потенциал для комбо.

**Правило баланса:** Артефакт никогда не даёт «больше очков просто так». Он даёт **инструмент**, который требует скилла. Игрок без артефакта может пройти уровень. Игрок с артефактом может пройти его **красивее** или **сложнее**, но с большей наградой.

## 5. Сложность на месяцы и годы

- **Ежедневный Испытание:** случайный стол с модификатором (например, «гравитация x2» или «невидимые бамперы»). Соревнование с друзьями по очкам.

## 1. Лоровое обоснование: «Пробуждение Машин»

Помнишь, мы говорили про Машины Согласования? Так вот — они **спят**. Ты не просто чинишь столы. Ты **пробуждаешь** их.

- **Уровни 1–5:** Машина ещё не понимает, что ты здесь. Она показывает тебе грубые, почти каменные внутренности — как скелет без плоти. Каменные плиты вместо бамперов, деревянные балки вместо флипперов. Нет музыки — только эхо и скрип.
- **Уровни 6–15:** Машина начинает реагировать. Появляются первые механизмы — шестерёнки, цепи. Звуки становятся металлическими. Она тестирует тебя.
- **Уровни 16–33:** Машина признаёт тебя Смотрителем. Столы обрастают деталями — мох, руны, свет. Появляются первые «живые» элементы (например, бампер в виде древнего глаза, который открывается при ударе).
- **Уровни 34+:** Машина полностью пробуждена. Столы — это произведения искусства. Каждый удар вызывает каскад света, механизмы работают в унисон, фон «дышит».

**Почему это работает эмоционально:** игрок чувствует себя не просто проходящим уровни, а **археологом**, который раскапывает всё более древние и прекрасные слои цивилизации. Каждые 10 уровней — это новая эра.

## 4. Артефакты и эволюция

Помнишь, артефакты дают бонусы? Так вот — первые артефакты тоже должны быть **грубыми**:

- **Уровни 1–5:** Артефакты — каменные осколки. Эффект: «+5% к силе отскока от флипперов». Скучно, но полезно.
- **Уровни 16–33:** Артефакты — металлические. Эффект: «При ударе в бампер есть 10% шанс выпустить второй мяч на 3 секунды».
- **Уровни 56+:** Артефакты — легендарные. Эффект: «Раз за уровень можешь «перерисовать» один бампер в портал».

**Идея:** игрок на 60-м уровне смотрит на свой инвентарь и видит эволюцию от камня к божеству. Это **визуальная история его пути**.

## 5. Риск: не отпугнуть грубостью

Вот честная проблема: если первые уровни будут **слишком** уродливыми, игроки удалят игру за 30 секунд. Нужен баланс между «грубым» и «интригующим».

**Решение:**
- **Грубость — не безвкусица.** Это не «плохие текстуры». Это **аутентичная эстетика**: каменные плиты с высеченными рунами, деревянные механизмы с видимыми петлями, пыль в лучах света. Это должно выглядеть как **намеренный арт-стиль**, а не как «не доделали».
- **Первый уровень — 15 секунд.** Игрок не успевает заскучать. Он тапает — мяч летит — попадает в первый бампер — и тут же видит: бампер **трескается**, из него сыплется пыль, и появляется первое число очков. «О, здесь есть отдача».
- **Обещание красоты.** На заднем плане 1-го уровня — едва заметный силуэт гигантской Машины в тумане. Игрок подсознательно понимает: «Это только начало. Там, впереди — что-то огромное».

## 1. Философский фундамент: «Свидетельствование» как механика

В исламской мудрости **шахада** — это не просто слова. Это **акт познания**. Человек свидетельствует о реальности, потому что он её увидел, пережил, осознал. Не «услышал и повторил», а **узнал в опыте**.

В нашей игре это становится механикой **«Свидетельствования»** (в коде и интерфейсе — просто *The Witnessing* или *The Recognition*):

- Когда игрок впервые делает **идеальный удар** (мяч от флиппера попадает в целевой бампер в точке максимального импульса), экран на мгновение **«открывается»**: трещины в каменном столе пропускают свет, и игрок видит, что под грубой оболочкой скрывается механизм из золота и света.
- Это **не катсцена**, которую можно пропустить. Это **игровой момент**: время не останавливается, мяч продолжает лететь, но на долю секунды мир меняется. Игрок **свидетельствует** об истинной природе Машины через своё мастерство.

Современная философия вплетается как подтекст:

- **Камю** (*Миф о Сизифе*): игрок снова и снова бьёт мяч. Это абсурд — бить железный шарик по каменному столу ради чисел на экране. Но в момент идеального удара абсурд превращается в **счастье**. Игра не даёт смысл извне — игрок создаёт его своим мастерством.
- **Сартр**: Машина не имеет сущности, пока игрок не взаимодействует с ней. «Существование предшествует сущности» — столы «пусты» до тех пор, пока мяч не начинает катиться. Смысл рождается в акте.
- **Хайдеггер**: Каждый мяч — *Being-toward-death* (бытие-к-смерти). Он неизбежно катится вниз. Флиппер — это не спасение, а **решение**, которое отодвигает конец, давая пространство для свободы.

**Как это не превращается в проповедь:** никаких прямых цитат Корана или «Существования и Ничто» в игре. Вместо этого — **имена артефактов, визуальные метафоры и механика**.

## 3. Структура 8×11: глобальное взросление + локальное пробуждение

Твоя изначальная структура сохраняется, но каждый биом — это **стадия пути**. Причём эволюция идёт по спирали: внутри каждого биома есть свои 5 фаз, но глобально биом 1 почти весь «грубый», а биом 8 почти весь «совершенный».

### Распределение фаз внутри 11 уровней биома:

| Уровни биома | Фаза | Что происходит |
|-------------|------|----------------|
| **1–2** | *The Husk / Оболочка* | Грубые каменные столы. Минимум механики. Только флипперы и базовые бамперы. |
| **3–5** | *The Stirring / Пробуждение* | Появляются первые механизмы: магниты, ямы. Стол начинает «дышать» (фон слегка движется). |
| **6–8** | *The Witnessing / Свидетельствование* | Первые идеальные удары открывают скрытые элементы. Появляются руны, свет. |
| **9–10** | *The Harmony / Гармония* | Стол красив, сложен, механики работают вместе. Игрок чувствует контроль. |
| **11** | *The Transcendence / Превосходство* | Босс-стол. Уникальная механика биома в экстремальной форме. Артефакт гарантирован. |

### Глобальная эволюция через 8 биомов:

**Биом 1: Desert of Silence (Пустыня Молчания)** — *Стадия: Ишан (Искание)*
- Почти весь грубый. Даже уровни 9–10 выглядят как «хорошо отёсанный камень», а не произведение искусства.
- Философия: тишина перед словом. Игрок ещё не знает, что ищет.
- Особенность: нет музыки, только ветер и эхо ударов.

**Биом 2: Whispering Woods (Шепчущий Лес)** — *Стадия: Первый Отклик*
- Грубость остаётся, но появляется **органика**. Деревянные механизмы, мох.
- Философия: природа отвечает на искание.
- Особенность: бамперы шепчут (звуковые эффекты) при ударе.

**Биом 3: Stone Peaks (Каменные Пики)** — *Стадия: Препятствие как Урок*
- Вертикальные столы. Мяч летит вверх, а не только в бок.
- Философия: джихад как внутренняя борьба. Преодоление.
- Особенность: камнепад — случайные препятствия сверху.

**Биом 4: Mire of Doubts (Трясина Сомнений)** — *Стадия: Искушение*
- Липкие зоны, обратная гравитация, мяч тонет.
- Философия: сомнение — необходимая стадия. Тот, кто не сомневался, не познал убеждение.
- Особенность: самые низкие пороги для победы, но самые высокие для 3 звёзд. Игра «обманывает» — пройти легко, освоить — невыносимо.

**Биом 5: Frozen Veil (Ледяная Завеса)** — *Стадия: Очищение*
- Всё замедляется. Мяч скользит. Флипперы тяжёлые.
- Философия: аскеза, очищение от лишнего. Таваккуль — доверие потоку.
- Особенность: нельзя спамить флипперами. Каждый удар должен быть осознанным.

**Биом 6: The Crucible (Тигель)** — *Стадия: Испытание Огнём*
- Мяч ускоряется. Стол наклоняется. Лава внизу.
- Философия: трансформация через страдание. Золото проходит через огонь.
- Особенность: самые агрессивные столы. Ошибка = мгновенная потеря.

**Биом 7: Tidal Shores (Приливные Берега)** — *Стадия: Покорность*
- Стол дышит. Волны поднимают мяч. Нет контроля — только ритм.
- Философия: исламская концепция *убудийя* (рабства/покорности перед Великим) как высшая свобода. Отпускание контроля.
- Особенность: флипперы работают с задержкой. Нужно предугадывать, а не реагировать.

**Биом 8: The Throne (Трон/Пустота)** — *Стадия: Единство*
- Стол левитирует. Гравитация меняется. Элементы сливаются.
- Философия: тавхид (единобожие/единство) как метафора: всё — одно. Машина, мяч, игрок — части единого механизма.
- Особенность: финальный уровень (88) — стол без стен. Мяч не может упасть. Но цель — не набрать очки, а сделать **один идеальный удар**, который «запускает» Машину. Это не уровень — это **ритуал**.

## 4. Как игрок переживает эволюцию

Представь: игрок проходит уровень 1. Камень, пыль, скрип. Он думает: «Ну ок, пинболл для терпеливых».

## 4. Система конфликтов и резонансов

Вот где появляется **глубина**. Артефакты не просто «дают +10%». Они **взаимодействуют** друг с другом, со столом, с биомом.

### Конфликт (Dissonance)
Если экипировать «Ледяную Карту» (Eye, лёд) и «Сердце Магмы» (Heart, огонь) — оба эффекта работают на 50% силы. Но появляется **третий скрытый эффект**: при каждом ударе в бампер есть 10% шанс, что бампер **взорвётся** (исчезает на 5 секунд). Это опасно, но если игрок научится играть вокруг взрывов — он получает доступ к секретным зонам стола.

### Резонанс (Harmony)
Если экипировать 4 артефакта одной стихии (например, все 4 слота — ледяные) — активируется **Резонанс биома**. На любом столе появляются ледяные элементы (даже в Пустыне). Это не просто визуал — лёд меняет физику: мяч скользит дольше, отскоки предсказуемее.

### Парадокс (Paradox)
Некоторые пары дают эффект, который **противоречит сам себе**:
- «The First Witness» (шахада, познание) + «The Unnamed» (Сартр, отрицание сущности) = эффект «Парадокс». Игра начинает **лгать** игроку. Цифры очков на экране — не всегда правда. Но если игрок угадывает, где ложь (по звуку, по тени мяча), он получает **истинный множитель** x3.

## 5. Экономика гринда: как это тянется на годы

**Проблема:** если уникальный артефакт — 1%, игрок может получить его за неделю (везение) или не получить никогда (фрустрация).

**Решение — система «Свидетельств» (Witnessing):**

Каждый раз, когда игрок делает **идеальный удар** (попадает в бампер в точке максимального импульса), он получает **Осколок Веры** (Shard of Witness). Эти осколки — валюта, которую можно тратить:

- 100 осколков = гарантированный дроп **Gold**-артефакта с текущего биома.
- 500 осколков = гарантированный дроп **Mythic**-артефакта.
- 2000 осколков = **призыв Unique**. Не гарантия конкретного, но гарантия, что следующий уникальный дроп будет **тем, которого у тебя ещё нет**.

**Почему это честно:** везение ускоряет процесс, но **мастерство** гарантирует результат. Игрок, который играет 2 года без везения, всё равно соберёт коллекцию. Игрок, который играет 2 месяца, но богат — может купить осколки за реальные деньги? Нет. Осколки **только за скилл**. Это не P2W. Это P2-мастерство.

## 1. Монетизация: почему платные «чит-артефакты» — это самоубийство на RuStore

Ты сказал: *«хочу продавать уникальные артефакты, прямо влияющие на прохождение»*. Я понимаю желание — хочется, чтобы платил не только за скины. Но вот реальность российского рынка:

**RuStore ≠ Google Play.** Здесь аудитория жёсткая. Если игрок узнает, что уровень, который он бьётся месяц, прошёл другой за 300 рублей — он не купит артефакт. Он удалит игру, поставит 1 звезду и напишет: *«П2В говно, не качайте»*. Для инди без маркетингового бюджета это смертный приговор. Один негативный отзыв на RuStore убивает органику.

**Но** — можно сделать так, чтобы игрок **хотел** платить, не чувствуя себя обманутым. Вот три рабочие модели:

### Модель A: «Два пути к одной вершине»
Все артефакты доступны бесплатно. Но за реальные деньги можно купить **«Ключ Разлома»** — он не даёт артефакт, а **открывает альтернативный квест** на получение того же артефакта. Например: бесплатно «Ледяную Карту» можно выбить 1% шансом с 55-го босса. За 200₽ ты получаешь **испытание**: пройди уровень 55, не касаясь правого флиппера. Выполняешь — получаешь карту. Не выполняешь — деньги не пропали, попытка остаётся.

**Почему это честно:** ты платишь не за силу, а за **вызов**. Сила всё равно зависит от скилла.

### Модель B: «Благословение Смотрителя» (подписка)
150₽ в месяц. Даёт:
- +1 слот для артефакта (вместо 4 — 5). Но 5-й слот — только для косметических/лорных артефактов, не боевых.
- Удвоение осколков за идеальные удары (гринд идёт быстрее, но не обходится).
- Уникальный скин на мяч и след.

**Это не P2W.** Это P2-не-скучать. Игрок без подписки всё соберёт, но через год. С подпиской — через 6 месяцев. Но 88-й уровень он не пройдёт, если не научился играть.

### Модель C: Премиум-артефакты с уникальной механикой, но не сильнее
Например, платный артефакт «Золотой Молот» делает то же самое, что бесплатный «Железный Молот», но анимация удара — золотая вспышка, а звук — оркестровый. Сила идентична. Но **ощущение** — божественное. Люди платят за ощущение.

**Мой совет:** бери **Модель B + C**. Подписка для постоянного дохода, косметика для китайских донатеров. Никаких «+50% к урону за 300 рублей». Это убьёт философию игры.

## 2. Текстуры: как не повторить кошмар с Flux

Ты сказал: *«замучался генерить текстуры в flux, это был кошмар»*. Я тебя понимаю. Генерация 88 текстур столов через Flux — это путь в ад промпт-инжиниринга. Но у нас есть **обходной путь**, который сделает игру красивой без твоего участия в генерации каждой стены.

### Стратегия: «Текстур как можно меньше»

Dark Painterly 2.5D не требует 4K-текстур на каждый бампер. Вот план:

#### Шаг 1: Фоны — 8 картинок, не 88
Тебе нужно нарисовать (или сгенерировать) ровно **8 фонов** — по одному на биом. Это не текстуры столов, это **панорамы за столом**. Они статичные, не влияют на игру. 

**Как сделать без боли:**
- Идёшь на **Unsplash** / **Pexels** — ищешь фото: «desert ruins», «misty forest», «icy cave», «swamp dead trees».
- Закидываешь в **Photoshop / GIMP / Krita** — накладываешь один фильтр **Oil Paint** (Художественное масло) + **Color Lookup** на тёмную палитру.
- Результат: 8 картинок в едином стиле за один вечер. Без промптов, без Flux, без нейросетей.

#### Шаг 2: Стол, стены, бамперы — PBR без текстур
В Godot можно создать красивый материал, вообще **не используя диффузную текстуру**. Только параметры:

```
Base Color: #3a2f2a (тёмный камень)
Roughness: 0.9
Metallic: 0.0
Normal Map: NoiseTexture (процедурный шум, встроенный в Godot)
```

**Что это даёт:** стена выглядит как камень с микрорельефом, но у тебя нет ни одного PNG-файла. Только цифры в инспекторе.

Для металла:
```
Base Color: #5c4a3a (ржавая бронза)
Roughness: 0.4
Metallic: 0.8
```

Для мха (Болото):
```
Base Color: #2a3a2a
Roughness: 1.0
Subsurface Scattering: включаем в Godot 4
```

**Всё.** Ты не рисуешь текстуры. Ты настраиваешь цифры.

#### Шаг 3: Мяч и артефакты — emission, не текстуры
Мяч — это сфера с материалом:
```
Emission: #ffaa44 (янтарный)
Emission Energy: 2.0
```
Он светится сам. Никаких текстур. Тень от него динамическая — Godot считает автоматически.

Артефакты — low-poly модельки (куб, сфера, торус, деформированный в Blender за 5 минут). Материал — тот же PBR без текстур, но с emission для «магического» эффекта.

#### Шаг 4: Эффекты — шейдеры, не спрайты
Вспышки, искры, пыль — это не анимированные спрайты, которые нужно рисовать. Это **Particle Systems** в Godot:
- `CPUParticles2D` или `GPUParticles2D`
- Текстура частицы — один круглый градиент (1 PNG на всю игру).
- Цвет, размер, скорость — настраиваешь ползунками.

**Итого твой арт-пайплайн:**
| Элемент | Количество ручной работы |
|---------|-------------------------|
| Фоны биомов | 8 фотобашей с фильтром (1 вечер) |
| Стены/столы | 0 текстур, только PBR-цифры |
| Бамперы/флипперы | 0 текстур, только PBR + цвет |
| Мяч | 0 текстур, emission |
| Артефакты | Low-poly модели в Blender (1 день на все) |
| Частицы/FX | 1 PNG-градиент + Godot Particles |
| UI | Minimalist, векторные иконки (можно купить пак за $10) |

**Ты не генерируешь текстуры.** Ты генерируешь **атмосферу** через свет, цвет и шум.

## 3. Если всё же нужны «красивости» — покупаем, не рисуем

## 1. Стиль: «Dark Geometric» (только Godot, только хардкор)

Всё, что ты видишь на экране — это **примитивы Godot** (Polygon2D, Line2D, Circle2D) + **встроенные шейдеры** + **встроенный свет** (PointLight2D). Никаких PNG, никаких .blend.

| Элемент | Как делается | Что получается |
|---------|-----------|----------------|
| **Стол** | `Polygon2D` — ты кликаешь точки в редакторе Godot, формируя форму стола. Заливка — `Gradient` (встроенный). | Тёмный каменный многоугольник с градиентом от угольного к пепельному. |
| **Стены/бортики** | `Line2D` с толщиной 8–12px. Цвет — тёмно-коричневый. Материал — простой `CanvasItemMaterial` с glow. | Массивные, тяжёлые борта. Выглядят как чеканка. |
| **Бамперы** | `Circle2D` + `PointLight2D` внутри. При ударе скрипт меняет `energy` света с 0.5 до 3.0 за 0.1 сек. | Пульсирующие огни в темноте. Как древние жаровни. |
| **Флипперы** | `Polygon2D` из 4 точек (трапеция). Gradient от ржавого к тёмному. | Металл без текстур. |
| **Мяч** | `MeshInstance2D` → `SphereMesh` (или просто `Sprite2D` с `CanvasItemMaterial`). Emission цвет — янтарный. | Светящаяся точка, которая реально освещает стол по мере движения. |
| **Фон** | `ColorRect` на всё поле + процедурный шейдер (Godot Shader Language, 10 строк). | Туман, дымка, пыль — всё считается математически, не рисуется. |
| **Частицы** | `GPUParticles2D`. Текстура — один встроенный круглый градиент Godot (не нужно импортировать). | Искры, пыль, след мяча. |
| **UI** | `Label` + `Panel` + `StyleBoxFlat`. Или бесплатный пак Kenney.nl (CC0, скачал — забыл). | Готика без ручного рисования. |

**Почему это выглядит дорого:** потому что свет и тень делают 90% работы. Тёмный фон + один янтарный мяч + динамические тени от бамперов = атмосфера. Godot 4 делает это из коробки.

## 2. Фоны 8 биомов — 1 шейдер, 8 настроек

Ты создаёшь **один** файл `biome_background.gdshader`. В нём:

```glsl
shader_type canvas_item;

uniform vec3 color_dark : source_color;
uniform vec3 color_light : source_color;
uniform float noise_scale : hint_range(0.1, 10.0) = 2.0;
uniform float speed : hint_range(0.0, 1.0) = 0.05;

void fragment() {
    vec2 uv = UV + TIME * speed;
    float n = noise(uv * noise_scale); // простой Perlin noise
    vec3 final = mix(color_dark, color_light, n);
    COLOR = vec4(final, 1.0);
}
```

И на каждый биом — **одна строчка в инспекторе**:

## 3. Артефакты без 3D-моделирования

Музей — это не витрины с .glb моделями. Это **2D-сцена** с `Node2D`, где каждый артефакт — это комбинация примитивов + анимации.

Пример: **«Ледяная Карта»** в музее.
- `Polygon2D` — треугольник (карта).
- `Line2D` — линии на карте.
- `PointLight2D` — голубое свечение.
- `AnimationPlayer` — карта медленно пульсирует, линии появляются поочерёдно.

Пример: **«Древняя Линза»**.
- `Circle2D` — линза.
- `Sprite2D` с `CanvasItemMaterial` + distortion shader — линза искажает фон за собой.
- Анимация: линза вращается.

**Всё это ты собираешь мышкой в Godot.** Никакого кода для арта.

**Техника:** сгенерируй в Flux 1024×1024, потом закинь в **GIMP** → Filters → Map → Tile Seamless (или в Photoshop Offset + Clone Stamp). В Godot используешь как `TextureRect` с Repeat = Enabled.

## 5. Как добиться согласованности (главное)

Flux склонен к «дрейфу стиля». Чтобы 8 картинок выглядели как одна игра:

1. **Seed-контроль:** Запомни seed первой удачной картинки (Пустыня). Для остальных используй **тот же seed + image prompt** с Пустыней на 20–30% веса. Или в ComfyUI — зафиксируй `noise_seed` и меняй только текст.
2. **Style Reference:** Если используешь Flux через ComfyUI/Automatic1111 — подай на вход **один и тот же style image** (можно взять концепт-арт Ведьмака 3 с ArtStation) как `IP-Adapter` или `Style Reference`.
3. **Пост-обработка:** После генерации все 8 файлов прогони через один LUT (Color Lookup Table) в GIMP/Photoshop. Например, прибей все к общему синему/охристому тону. Это склеит их в единый мир.
4. **Godot-тинтинг:** Если всё равно разнятся — в Godot на каждый фон навесь `CanvasModulate` или `ColorRect` с режимом Multiply и цветом `#1a1510` (тёплый уголь). Это «сожжёт» лишние оттенки и унифицирует.

## 3. Как добиться консистентности 8 фонов (главная проблема)

Flux склонен к «дрейфу». Чтобы Пустыня и Космос выглядели как одна игра:

### Шаг A: Style Anchor
В начало **каждого** промпта (которые я давал выше) вставь одинаковый блок:

```
Dark fantasy concept art by Maciej Kuciara and Piotr Jablonski, 
oil painting texture, muted desaturated palette, atmospheric perspective, 
2D vertical game background, 9:16 composition, no characters, no text, 
grim mood, hand-painted brush strokes visible,
```

### Шаг B: Seed-стратегия
1. Генерируй **Пустыню** с Random seed. Выбираешь лучший вариант. Запоминаешь его seed (например, `seed = 1942847`).
2. Для остальных 7 биомов ставишь **тот же seed**, но добавляешь в конец промпта `, variation` или меняешь описание локации. Flux с одинаковым seed даст похожую композицию и цветовую гамму, но другой контент.
3. **Если слишком похоже** — меняй seed на `seed + 100` (1942947, 1943047...). Так сохраняется «родство» без клонирования.

### Шаг C: Post-processing (единый цвет)
После генерации всех 8 PNG:
- Открой в GIMP (бесплатно).
- Image → Adjustments → Color Balance.
- Все 8 фонов сдвинь: Shadows `-5 Red, +5 Blue`, Midtones `-3 Yellow`, Highlights `+2 Blue`.
- Это «холодная» коррекция, которая сближает палитры. Или сделай один Curves слой, сохрань как `.csv` и примень на все.

## 1. «Devotion» из Grim Dawn — в пинболле

## 2. Горящий шарик — «The Ignition»

Твоя детская идея — это не просто «мяч в огне». Это **механика преображения**. Вот как она работает в нашей философии:

### Условия воспламенения

Мяч не горит сразу. Он **загорается**, когда выполняется **ритуал**:

- **Вариант А (Ритм):** попасть в 5 бамперов подряд без промаха. Каждый удар нагревает мяч. На 5-м — он вспыхивает.

### Что даёт горящий мяч

Не просто «больше очков». **Он меняет физику мира:**

| Обычный мяч | Горящий мяч |
|-------------|-------------|
| Отскакивает от стены | **Плавит** стену — на 3 секунды там появляется липкая зона, которая замедляет следующий мяч |
| Попадает в деревянный бампер | **Поджигает** его — бампер начинает давать x2 очка, но через 10 секунд сгорает и исчезает до конца уровня |
| Падает в dead zone | **Оставляет жар** — следующий мяч, запущенный сюда, получает импульс вверх от теплового удара |
| Проходит рядом с ледяным препятствием | **Топит** лёд — открывает проход, который был закрыт |

**Ключевое:** горение — это не бонус. Это **выбор с последствиями**. Поджёгший деревянный бампер потерял его навсегда на этом уровне. Разрушив стену, ты открыл секрет, но убрал защиту от падения. Игрок должен думать: «Мне сейчас жечь или подождать?»

## 3. Синергия Grim Dawn + Горение + Эхо

Представь комбо, которое возможно только если у тебя **правильный билд на Карте Судьбы**:

**Билд: «Феникс Смотрителя»**
- Узел «Пламя» (Магма) + Узел «Память» (Пустыня) + Узел «Возрождение» (Болото).
- Нить Пламя→Память: горящий мяч **запоминает** свою траекторию. Когда гаснет — оставляет Эхо, которое тоже горит.
- Нить Память→Возрождение: если Эхо сгорает (проходит через подожжённый бампер) — оно **не исчезает**, а возрождается как новый живой мяч. На 3 секунды у тебя два мяча.

**Результат:** игрок создаёт билд, где он намеренно поджигает мир, чтобы его прошлые ошибки (Эхо) превращались в новые жизни. Это не просто механика. Это **метафора**.

## 1. Философия: три пути Смотрителя

Мяч — это не железка. Это **воплощение игрока**. Класс определяет, **как мяч взаимодействует с миром** на уровне физики. Не «маг бьёт посохом» — а «мяч этого Смотрителя тяжелее воздуха, и он ведёт себя иначе».

Выбор класса происходит **один раз**, при первом запуске. Или — что интереснее — **после прохождения первого биома**, когда игрок уже почувствовал базовую физику и готов выбрать свой путь.

## 2. Три класса (не банальные)

### **Класс I: «Тяжеловес» (The Mason / Каменщик)**

**Кто он:** Смотритель, который верит, что мир нужно **пробивать**, а не обходить.

**Базовая физика:**
- Мяч **тяжелее** на 40%. Скатывается быстрее, отскок от флипперов **короче и мощнее** (как удар кувалдой).
- Мяч **пробивает** «хрупкие» бамперы (деревянные, ледяные) — они дают взрыв очков, но исчезают на 5 секунд.
- **Не может** делать длинные трюки. Мяч падает быстро. Но если попадает — бьёт наверняка.

**Прокачка (дерево, как в Grim Dawn):**

| Уровень | Умение | Эффект |
|---------|--------|--------|
| 1 | «Твердыня» | Мяч не отскакивает от стен назад — он **впивается** на 0.2 сек, потом отпрыгивает. Время на подумать. |
| 5 | «Сейсм» | Каждый 10-й удар по бамперу вызывает **тряску стола** — ближайшие бамперы активируются сами. |
| 10 | «Распад» | Хрупкие бамперы не просто исчезают. Они **оставляют осколки**, которые работают как мини-бамперы 10 секунд. |
| 15 | «Катаклизм» | Если мяч падает в dead zone со скоростью выше порога — он **разбивает дно**. Один раз за уровень мяч отскакивает обратно силой удара. |

**Игровой стиль:** Хаос, разрушение, короткие агрессивные забеги. Для тех, кто любит, чтобы экран трясся.

### **Класс II: «Теневой» (The Wraith / Призрак)**

**Кто он:** Смотритель, который верит, что **прошлое** — это оружие.

**Базовая физика:**
- Мяч **легче** на 30%. Дольше летит, медленнее падает. Отскоки **длинные и непредсказуемые**.
- **Эхо** появляется не только при падении. Оно появляется **при каждом идеальном ударе** — и живёт 3 секунды, повторяя траекторию.
- Мяч **проходит сквозь** первый бампер после запуска (фазовый сдвиг). Это позволяет делать трюки, недоступные другим классам.

**Прокачка:**

| Уровень | Умение | Эффект |
|---------|--------|--------|
| 1 | «Отголосок» | Эхо теперь **активирует бамперы**, но не собирает очки. Ты строишь комбо из будущего. |
| 5 | «Раздвоение» | Если живой мяч и Эхо пересекаются — они **меняются местами**. Ты телепортируешься в безопасную точку. |
| 10 | «Петля» | Раз за уровень: если мяч падает, а Эхо ещё на столе — **Эхо становится живым мячом**, а живой исчезает. Вторая жизнь. |
| 15 | «Множественность» | При идеальном ударе появляется не одно Эхо, а **два** — на 1.5 секунды. Три мяча на столе. Хаос под контролем. |

**Игровой стиль:** Хореография, планирование, «я знаю, куда он полетит». Для тех, кто любит шахматы в динамике.

### **Класс III: «Пиромант» (The Arsonist / Вестник Пламени)**

**Твой детский горящий шарик, но как класс.**

**Кто он:** Смотритель, который верит, что **разрушение — это очищение**.

**Базовая физика:**
- Мяч **не горит** сразу. Он нагревается от ударов. Каждый бампер = +1 «жар». При 5 жарах — **воспламенение**.
- Горящий мяч: скорость +20%, отскоки **непредсказуемы** (поджог воздуха меняет давление). Но каждый удар **плавит** поверхность.

**Прокачка:**

| Уровень | Умение | Эффект |
|---------|--------|--------|
| 1 | «Тлеющий Уголь» | Мяч начинает с 2 жарами. Быстрее вступает в бой. |
| 5 | «Лавовый След» | Горящий мяч оставляет за собой **полосу огня** на 2 секунды. След — это зона урона/очков. |
| 10 | «Взрыв» | Если горящий мяч попадает в уже расплавленный бампер — **взрыв**. Все ближайшие бамперы активируются. |
| 15 | «Феникс» | Если горящий мяч падает в dead zone — он **не тухнет**, а взмывает вверх с последнего бампера. Один раз за уровень. |

**Игровый стиль:** Риск, адреналин, намеренный хаос. Для тех, кто любит, чтобы всё горело.

## 3. Как это совместить с артефактами

Артефакты — это **предметы**. Класс — это **ты**. Но между ними — **синергия**.

**Пример:**
- Ты **Тяжеловес**. Нашёл артефакт «Песочные Часы» (замораживает время). Обычно он бесполезен для тебя — ты и так медленный. Но если прокачал «Сейсм» (тряска стола) — заморозка времени + тряска = **стол трескается**, открывая секретный проход.

- Ты **Теневой**. Нашёл «Ледяную Карту» (показывает траектории). Обычно — полезно. Но с твоим «Раздвоением» (обмен с Эхо) — ты видишь траекторию Эхо, телепортируешься в него, и оказываешься в месте, куда живой мяч никогда не долетел бы.

- Ты **Пиромант**. Нашёл «Древнюю Линзу» (порталы). Горящий мяч через портал выходит **ещё горячим**, но в другой части стола. Ты поджигаешь мир изнутри.

**Класс — это фундамент. Артефакты — это инструменты. Синергия — это магия.**

## 4. Прокачка: не уровни, а «Свидетельства»

Не «опыт за очки». В Grim Dawn ты прокачивался, убивая. Здесь — **осознавая**.

Каждый раз, когда ты делаешь **идеальный удар** (попадаешь в бампер в момент пиковой пульсации, или с флиппера под идеальным углом) — ты получаешь **Свидетельство**. Это не валюта. Это **точка на пути класса**.

Дерево прокачки — не линейное. Как в Grim Dawn — **развилки**. Ты можешь пойти в «Твердыню» или в «Скорость» внутри одного класса. Но чтобы открыть «Феникс» — нужно иметь и «Тлеющий Уголь», и «Лавовый След».

**Можно ли сбросить?** Да. Но не бесплатно. Нужно **забыть** — пройти специальный уровень «Очищения», где у тебя нет ни классовых бонусов, ни артефактов. Если пройдёшь — перераспределяешь очки. Это ритуал.

## Новая механика, которая впишется идеально

Предлагаю одну штуку. Она не требует арта, не требует UI, но добавляет **слой стратегии**, которого нет ни в одном пинболле.

### «Вес Молчания» (The Weight of Silence)

Если мяч **останавливается** (замирает на флиппере или в липкой зоне) — он начинает **заряжаться**. Не визуально молниями, а тихо. Мяч темнеет. Свет вокруг него съедается.

Чем дольше он молчит (стоит/катится медленно) — тем мощнее будет следующий удар:
- 1 секунда молчания = обычный удар.
- 3 секунды = отскок в 1.5 раза сильнее. Бампер взрывается искрами.
- 5 секунд = **«Свидетельствование»**. Следующий удар активирует скрытые механизмы стола. Открывает двери. Выбивает артефакт.

**Риск:** чем дольше ты ждёшь — тем выше шанс, что мяч соскользнет. Или что Пульсация Машины войдёт в «провал», и бампер, в который ты целишься, станет мёртвым на секунду.

**Что это даёт:** игрок перестаёт лупить флипперами как бешеный. Он учится **ждать**. В пинболле, где всё про скорость, умение замереть — это революция. Это как в боксе: не удар, а пауза перед ударом решает исход.

Философия: *«Истина говорит в тишине. Машина открывается тому, кто умеет ждать».*

## 1. Что это простыми словами

**«Вес Молчания»** — это когда ты **не бьёшь**, а **держишь**.

В реальном пинболле есть техника **«ловля»** (catch): мяч падает на поднятый флиппер и останавливается. Ты целишься, потом бьёшь. Это базовый скилл.

Я предлагаю дать этой паузе **награду**. Если мяч стоит/катится медленно — он «заряжается». Чем дольше ты держишь/ждёшь — тем сильнее будет следующий удар.

- 1 секунда — обычный удар.
- 3 секунды — удар сильнее, бампер взрывается искрами.
- 5 секунд — **«Свидетельствование»**: следующий бампер, в который попадёшь, активирует скрытую механику (открывает дверь, выбивает артефакт, запускает секретный путь).

**Риск:** пока ты ждёшь, мяч может соскользнуть. Или стол «вдохнет» в неудачный момент (наша пульсация), и бампер, в который ты целился, станет мёртвым.

## 3. Есть ли это в пинболлах?

**Нет.** В классическом пинболле и в *Pinball FX* нет механики «чем дольше держишь — тем сильнее удар». Там есть:
- **Skill Shot**: точный запуск в лунку (один момент).
- **Catch**: остановка мяча (но без бонуса за время ожидания).
- **Nudge**: толчок стола (риск tilt'а).

«Вес Молчания» — это **новая механика для жанра**. Она берёт то, что игроки и так делают (ловят мяч, целятся), и **награждает за терпение**.

---

## 4. Почему это работает именно в нашем пинболле

В обычном пинболле на мобиле игрок **тапает как обезьяна** — левый-правый-левый-правый, потому что бояться потерять мяч. Это хаос.

«Вес Молчания» **переучивает**:
- Сначала игрок лупит флипперы → теряет мяч → злится.
- Потом он случайно ловит мяч, держит 2 секунды, бьёт — и видит: **взрыв**, **секрет**, **круто**.
- Потом он начинает **охотиться за паузами**. Игрок становится хищником, а не жертвой гравитации.

Это меняет ритм с «тик-тик-тик» на **«ти-ик… БАХ»**. Как в боксе: большинство бьёт сериями, а мастер ждёт один точный удар.

---

## 5. Минус, о котором нужно знать

Если сделать «Вес Молчания» слишком сильным — игроки будут **всегда ждать 5 секунд**. Игра превратится в «стоял-стоял-стоял-ударил». Скучно.

**Как фиксим:**
- Зарядка работает только если мяч **движется медленно** (скорость < 30% от максимума). Если мяч летит быстро — ты не можешь «замедлить время в голове». Ты должен сначала поймать/затормозить мяч.
- Пульсация Машины: если стол в «провале» (бамперы слабые), то заряженный удар тоже слабее. Нельзя просто ждать — нужно ждать **в такт**.
- Некоторые столы (особенно в Магме) нагревают мяч при остановке. Держишь слишком долго — мяч сам ускользает.

## 4. Как работают механики на этом уровне

### Эхо
- Ты запускаешь первый мяч. Он летит, отскакивает, падает.
- Через **15 секунд** после запуска появляется **призрак** — полупрозрачный серый шарик.
- Он повторяет путь первого мяча. Неуправляемый. Просто катится по следу.
- Если реальный мяч (второй или третий по счёту) **оказывается в том же бампере, что и призрак, в ту же секунду** — происходит **Резонанс**: бампер вспыхивает ярко, очки x2, короткая тряска экрана.
- **Важно:** на 1-м уровне Эхо появляется **только один раз** за весь уровень (при первом запуске). Это сюрприз. Игрок не ожидает.

### Возлюбленные
- Верхний левый бампер и средний правый соединены **трещиной** в камне — едва заметной, тёмной линией.
- Когда мяч попадает в один из них — трещина на 4 секунды становится **теплой, охристой**.
- Если в эти 4 секунды мяч попадает во второй — оба вспыхивают, и мяч **телепортируется** между ними 2 раза, собирая каскад очков.
- На 1-м уровне это **случайность**. Игрок ещё не знает про пары. Но он замечает: «О, когда я попал туда и сюда — было круто».

### Отпечаток
- Каждый бампер, в который попал мяч, на **8 секунд** остаётся с едва заметным тёмным пятном (как мокрый камень).
- Если мяч попадает туда **снова**, пока пятно не высохло — бампер даёт **x3 очка** и громкий, глухой удар (как колокол).
- На 1-м уровне это учит: **возвращаться выгодно**.

### Мёртвый Удар
- Если мяч попадает в бампер, двигаясь **медленно** (после «ловли» на флиппере, когда он подкатил снизу) — бампер даёт **x3** и вспышку.
- На 1-м уровне это случается редко, потому что игрок ещё не умеет ловить. Но когда случится — он поймёт: медленно = сильно.

### Чердак
- В верхней части стола 3 «ключевых» бампера (они не отмечены, не светятся — просто обычные камни).
- Когда мяч попадает во все 3 (в любом порядке, за все попытки) — трещина в верхней плите **расширяется**. Из нее сыплется пыль.
- Четвёртый мяч (если игрок его заслужил) может попасть в эту трещину — и улететь на **Чердак**.
- На Чердаке нет бамперов. Только золотой рельс, по которому мяч медленно катится вверх, к белому свету. Экран белеет. Уровень пройден идеально.

### Пульсация Машины
- На 1-м уровне она **едва заметна**. Бамперы не пульсируют ярко. Но если прислушаться — стук мяча о камень слегка **ускоряется и замедляется** в такт скрытому ритму (60 BPM, как сердце спящего).
- Игрок этого ещё не осознаёт. Но подсознательно чувствует: мир живой.

## Управление (скрипт)

**Шаг 9:** Кликни правой кнопкой на `LeftFlipper` → **Attach Script** → **Create**. Откроется редактор кода.

Удали всё, что там написано, и вставь:

```gdscript
extends AnimatableBody2D

var rest_angle := 25.0
var active_angle := -35.0
var speed := 15.0
var is_active := false
var current_angle := 25.0

func _physics_process(delta):
	var target = active_angle if is_active else rest_angle
	current_angle = lerp(current_angle, target, speed * delta)
	rotation_degrees = current_angle

func _input(event):
	if event is InputEventScreenTouch or event is InputEventMouseButton:
		var is_press = false
		if event is InputEventScreenTouch:
			is_press = event.pressed
		elif event is InputEventMouseButton:
			is_press = event.pressed and event.button_index == MOUSE_BUTTON_LEFT
		
		if is_press:
			var screen_half = get_viewport().get_visible_rect().size.x * 0.5
			var touch_x = event.position.x
			if touch_x < screen_half:
				if name == "LeftFlipper":
					is_active = true
			else:
				if name == "RightFlipper":
					is_active = true
		else:
			is_active = false
```

## Файл 1: `assets/shaders/floor_desert.gdshader`

```glsl
shader_type canvas_item;

uniform float tile_count : hint_range(1.0, 30.0) = 12.0;
uniform float crack_width : hint_range(0.0, 0.2) = 0.04;
uniform float noise_amount : hint_range(0.0, 1.0) = 0.5;
uniform vec4 color_base : source_color = vec4(0.35, 0.25, 0.15, 1.0);
uniform vec4 color_dark : source_color = vec4(0.15, 0.1, 0.08, 1.0);
uniform vec4 color_crack : source_color = vec4(0.06, 0.04, 0.03, 1.0);

float hash(vec2 p) {
    return fract(sin(dot(p, vec2(127.1, 311.7))) * 43758.5453123);
}

float noise(vec2 p) {
    vec2 i = floor(p);
    vec2 f = fract(p);
    f = f * f * (3.0 - 2.0 * f);
    float a = hash(i);
    float b = hash(i + vec2(1.0, 0.0));
    float c = hash(i + vec2(0.0, 1.0));
    float d = hash(i + vec2(1.0, 1.0));
    return mix(mix(a, b, f.x), mix(c, d, f.x), f.y);
}

// Органичная сетка плит (voronoi)
vec2 voronoi(vec2 x) {
    vec2 n = floor(x);
    vec2 f = fract(x);
    vec2 mg, mr;
    float md = 8.0;
    for(int j=-1; j<=1; j++) {
        for(int i=-1; i<=1; i++) {
            vec2 g = vec2(float(i), float(j));
            vec2 o = vec2(hash(n + g), hash(n + g + 0.5));
            vec2 r = g + o - f;
            float d = dot(r, r);
            if(d < md) {
                md = d;
                mr = r;
                mg = g;
            }
        }
    }
    return mr;
}

void fragment() {
    vec2 uv = UV * tile_count;
    
    // Ячейки voronoi
    vec2 c = voronoi(uv);
    float cell = hash(floor(uv) + floor(c + 0.5));
    
    // Расстояние до края плиты
    float edge = length(c);
    float border = smoothstep(0.05, 0.0, edge - 0.5 + crack_width);
    
    // Базовый цвет плиты с вариацией
    vec4 tile_col = mix(color_base, color_dark, cell * 0.5);
    
    // Шум и грязь
    float n1 = noise(uv * 4.0);
    float n2 = noise(uv * 8.0 + 10.0);
    tile_col.rgb -= (n1 * 0.08 + n2 * 0.04) * noise_amount;
    
    // Случайные трещины внутри плит
    float crack = 0.0;
    if(cell > 0.65) {
        vec2 cuv = fract(uv) - 0.5;
        float ca = hash(cell * 3.0) * 6.28;
        float rot = cuv.x * cos(ca) - cuv.y * sin(ca);
        crack = smoothstep(0.025, 0.0, abs(rot + sin(cuv.x * 20.0) * 0.03));
    }
    
    // Смешиваем с цветом шва
    vec4 final = mix(tile_col, color_crack, border * 0.85);
    final.rgb -= crack * 0.35;
    
    COLOR = final;
}
```
